import { useInfiniteQuery } from '@tanstack/react-query';
import { useTranslation } from 'react-i18next';
import { api } from '../../api/endpoints';
import type { Field, SettingsEditor } from './settings';

const fields:Field[]=['inboundRingTimeoutSec','outboundUserAnswerTimeoutSec','outboundCalleeAnswerTimeoutSec','outboundCallbackRingTimeoutSec','alertPushIntervalSec','earlyMediaThresholdDbfs'];
export function MoreSettings({editor}:{editor:SettingsEditor}){
 const {t}=useTranslation(); const {settings,drafts,busy,blocked,edit,resolve,save}=editor;
 const devices=useInfiniteQuery({queryKey:['devices'],queryFn:({pageParam,signal})=>api.devices(pageParam,signal),initialPageParam:undefined as string|undefined,getNextPageParam:page=>page.nextCursor??undefined});
 const conflict=(field:Field)=>drafts[field]?.conflict&&<div role="alert"><p>{t('settingsPage.draftConflict',{field:t('moreSettings.'+field)})}</p><button type="button" disabled={busy||blocked} onClick={()=>resolve(field,false)}>{t('settingsPage.useLatest')}</button><button type="button" disabled={busy||blocked} onClick={()=>resolve(field,true)}>{t('settingsPage.keepMine')}</button></div>;
 return <><form className="panel" onSubmit={e=>{e.preventDefault();save(['defaultDeviceId']);}}><h2>{t('moreSettings.defaultDevice')}</h2>{devices.isError?<p role="alert">{t('loadError')}</p>:<><label className="field">{t('moreSettings.defaultDevice')}<select disabled={busy} value={String(drafts.defaultDeviceId?.value??settings.defaultDeviceId??'')} onChange={e=>edit('defaultDeviceId',e.target.value)}><option value="">{t('moreSettings.none')}</option>{devices.data?.pages.flatMap(p=>p.data).map(d=><option key={d.id} value={d.id}>{d.name}</option>)}</select></label>{devices.hasNextPage&&<button type="button" onClick={()=>void devices.fetchNextPage()}>{t('loadMore')}</button>}{conflict('defaultDeviceId')}<button className="secondary" disabled={busy||blocked||devices.isPending||!drafts.defaultDeviceId||drafts.defaultDeviceId.conflict}>{t('action.save')}</button></>}</form><form className="panel" onSubmit={e=>{e.preventDefault();save(fields);}}><h2>{t('moreSettings.callTiming')}</h2>{fields.map(field=><div key={field}><label className="field">{t('moreSettings.'+field)}<input type="number" step={field==='earlyMediaThresholdDbfs'?'any':undefined} required disabled={busy} value={String(drafts[field]?.value??settings[field])} onChange={e=>edit(field,e.target.value)}/></label>{conflict(field)}</div>)}<button className="secondary" disabled={busy||blocked||!fields.some(f=>drafts[f])||fields.some(f=>drafts[f]?.conflict)}>{t('action.save')}</button></form></>;
}
